user nginx;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;
worker_processes auto;
worker_rlimit_nofile 33282;

events {
  worker_connections 1024;
}

http {
  include /etc/nginx/mime.types;
  default_type application/octet-stream;

  log_format healthd '$msec"$uri"'
                    '$status"$request_time"$upstream_response_time"'
                    '$http_x_forwarded_for';

  # Your custom health check bypass (handles /health directly)
  location = /health {
    access_log off;
    default_type text/plain;
    return 200 "OK\n";
  }

  # Your optimizations
  types_hash_max_size 2048;
  types_hash_bucket_size 64;
  client_max_body_size 20M;

  # Pull in Elastic Beanstalk's essential configs (enhanced health, proxy, etc.)
  include conf.d/elasticbeanstalk/*.conf;

  # Default server block (proxies everything else to your app on 5000)
  server {
    listen 80;
    gzip on;
    gzip_proxied any;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml+rss text/javascript;

    location / {
      proxy_pass http://127.0.0.1:5000;
      proxy_set_header Connection $http_connection;
      proxy_set_header Upgrade $http_upgrade;
      proxy_set_header Host $host;
      proxy_set_header X-Real-IP $remote_addr;
      proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
      proxy_set_header X-Forwarded-Proto $scheme;
      proxy_http_version 1.1;
      proxy_buffering on;
    }

    access_log /var/log/nginx/access.log healthd;
  }
}
